        this.notes.Clear(); // Dodano

interface IBoxIterator {
    Product First();
    Product Next();
    bool IsDone { get; }
    Product Current { get; }
}

interface IBoxCollection {
    IBoxIterator GetIterator();
}

class Box : IBoxCollection
{
    private List<Product> products;
    
    public Box() { this.products = new List<Product>(); }
    public Box(List<Product> products) { this.products = new List<Product>(products.ToArray()); }
    
    public void AddProduct(Product product) { this.products.Add(product); }
    public void RemoveProduct(Product product) { this.products.Remove(product); }
    public int Count { get { return this.products.Count; } }
    
    // Potreban indekser za iterator
    public Product this[int index] { get { return this.products[index]; } }

    public IBoxIterator GetIterator() {
        return new BoxIterator(this);
    }
}

class BoxIterator : IBoxIterator
{
    private Box box;
    private int currentPosition;

    public BoxIterator(Box box) {
        this.box = box;
        this.currentPosition = 0;
    }

    public bool IsDone { get { return this.currentPosition >= this.box.Count; } }
    public Product Current { get { return this.box[this.currentPosition]; } }

    public Product First() { return this.box[0]; }

    public Product Next() {
        this.currentPosition++;
        if (this.IsDone) return null;
        else return this.box[this.currentPosition];
    }
}

{
    ToDoItem zadatak = new ToDoItem("Kupovina", "Kupi kruh i mlijeko", DateTime.Now.AddDays(1));

    // Promijeni stanje i spremi

    zadatak.ChangeTask("Kupi sve iz trgovine");

    // Vrati na zadnje spremljeno stanje (undo)
    zadatak.RestoreState(careTaker.SavedStates[1]);
    Console.WriteLine("--- Nakon vraćanja ---");
    Console.WriteLine(zadatak.ToString());
}

using System.Linq;

class StringDigitChecker : StringChecker {
    protected override bool PerformCheck(string stringToCheck) {
        return stringToCheck.Any(char.IsDigit);
    }
}

class StringLengthChecker : StringChecker {
    protected override bool PerformCheck(string stringToCheck) {
        return stringToCheck.Length >= 8; // Minimalno 8 znakova
    }
}

class StringUpperCaseChecker : StringChecker {
    protected override bool PerformCheck(string stringToCheck) {
        return stringToCheck.Any(char.IsUpper);
    }
}

class StringLowerCaseChecker : StringChecker {
    protected override bool PerformCheck(string stringToCheck) {
        return stringToCheck.Any(char.IsLower);
    }
}














class PasswordValidator
{
    private StringChecker firstChecker;
    private StringChecker currentChecker;

    public PasswordValidator(StringChecker firstChecker) {
        this.firstChecker = firstChecker;
        this.currentChecker = firstChecker;
    }

    public void AddChecker(StringChecker nextChecker) {
        this.currentChecker.SetNext(nextChecker);
        this.currentChecker = nextChecker;
    }

    public bool Validate(string password) {
        return this.firstChecker.Check(password);
    }
}

// TESTIRANJE (Zadaci 6 i 7)
public static void TestZadatak7()
{
    PasswordValidator validator = new PasswordValidator(new StringLengthChecker());
    validator.AddChecker(new StringDigitChecker());
    validator.AddChecker(new StringUpperCaseChecker());
    validator.AddChecker(new StringLowerCaseChecker());

    bool isStrong = validator.Validate("Slabalozinka");
    Console.WriteLine($"'Slabalozinka' validacija: {isStrong}"); // false

    bool isStrong2 = validator.Validate("Jaka1Lozinka");
    Console.WriteLine($"'Jaka1Lozinka' validacija: {isStrong2}"); // true
}
